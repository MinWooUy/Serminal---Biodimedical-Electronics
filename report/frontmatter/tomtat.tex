\chapter*{Tóm tắt}
\addcontentsline{toc}{chapter}{Tóm tắt}

[Viết liền mạch 200--300 từ, văn phong học thuật, khách quan, không gạch đầu dòng.]

\begin{goiy}[Dàn ý tóm tắt (5 ý, theo mẫu của khoa)]
  \begin{enumerate}
    \item \textbf{Bối cảnh và vấn đề (1--3 câu):} theo dõi ECG liên tục cần phân loại loạn nhịp nhanh, chính xác, tiết kiệm tài nguyên; khoảng trống: chạy CNN số nguyên trọn vẹn trên FPGA nhỏ và kiểm chứng được từng bit.
    \item \textbf{Mục tiêu (1 câu):} thiết kế và kiểm chứng bộ tăng tốc CNN1D int8 phân loại nhịp N/S/V/F trên DE10-Lite (MAX 10).
    \item \textbf{Phương pháp (1--3 câu):} pipeline tiền xử lý số nguyên; CNN Lite 23~KB int8; accelerator mảng PE $7\times7$ điều khiển qua UART bằng bộ lệnh CONFIG/RUN; một mảng PE dùng chung cho conv và FC.
    \item \textbf{Kết quả chính (1--2 câu):} macro-F1 trên DS2 $\approx 0.603$; 1\,306/1\,306 nhịp khớp bit-exact với golden; 83\,\% LE, 50~MHz; 41.17~ms/nhịp tính toán. (Đối chiếu lại số liệu cuối cùng trước khi nộp.)
    \item \textbf{Kết luận (1--2 câu):} ý nghĩa; hạn chế (băng thông nạp, lớp S) và khả năng mở rộng.
  \end{enumerate}
\end{goiy}

\medskip
\noindent\textbf{Từ khoá:} ECG; phân loại nhịp tim; CNN 1D; lượng tử hoá int8; FPGA; bộ tăng tốc; UART.
