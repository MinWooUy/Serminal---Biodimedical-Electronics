\section{Mục tiêu và câu hỏi nghiên cứu}\label{sec:mucTieu}
\begin{goiy}
  \begin{itemize}
    \item \textbf{Câu hỏi nghiên cứu:} có thể chạy trọn một CNN 1D int8 phân loại nhịp trên FPGA nhỏ (MAX~10 10M50) mà kết quả khớp từng bit với mô hình số nguyên phần mềm hay không?
    \item \textbf{Mục tiêu đo được:} (i) macro-F1 trên DS2; (ii) bit-exact trên các đoạn golden; (iii) đạt 50~MHz và vừa tài nguyên chip.
    \item Mục tiêu phụ: giao thức Host--FPGA tin cậy; công cụ quan sát trực tiếp trên board.
  \end{itemize}
\end{goiy}
